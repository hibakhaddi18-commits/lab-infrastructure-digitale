! ============================================================
! CONFIGURATION DE BASE — EXEMPLE GENERIQUE A ADAPTER
! Ne pas appliquer telle quelle : adapter les noms, les interfaces
! et les adresses IP, et CHANGER les mots de passe d'exemple
! avant tout usage reel. Base en attente d'adaptation aux
! maquettes Packet Tracer de l'auteure.
! ============================================================
! ============================================================
! Configuration de base - Routeur Cisco
! Lab Infrastructure Digitale - Hiba Khaddi
! A adapter avant utilisation : nom, domaine, interface et IP
! ============================================================

enable
configure terminal

! Nom de l'equipement et domaine pour SSH
hostname R1
ip domain-name isfo.com

! Securite de base
enable secret cisco123
service password-encryption
no ip domain-lookup
security passwords min-length 8

! Compte administrateur local
username admin privilege 15 secret admin123

! Banniere d'avertissement
banner motd # Acces reserve au personnel autorise #

! Horloge et journalisation
clock timezone CET 1
logging synchronous

! Interface LAN - adapter le nom de l'interface a ton modele
interface gigabitEthernet0/0
 description Lien vers le LAN
 ip address 192.168.1.1 255.255.255.0
 no shutdown
exit

! Ligne console
line console 0
 password cisco123
 login
 logging synchronous
exit

! Generer la cle RSA necessaire pour SSH
crypto key generate rsa modulus 1024

! Lignes VTY : acces Telnet desactive, SSH uniquement
line vty 0 4
 login local
 transport input ssh
exit

end

! Sauvegarder la configuration dans la NVRAM
write memory
