! ============================================================
! FICHIER A ADAPTER - EXEMPLE DE CONFIGURATION
! Les adresses IP, noms et mots de passe ci-dessous sont des
! EXEMPLES tires de la topologie (capture d'ecran).
! Verifie et adapte-les a ton adressage reel avant usage.
! Ne pas appliquer tel quel sur un reseau de production.
! ============================================================
!
! Topologie : Router0 -- Router1, Switch0 (LAN0), Switch1 (LAN1), PC0
! Adressage exemple :
!   Lien R0-R1  : 192.168.12.0/30  (R0 .1, R1 .2)
!   LAN0 (Switch0) : 192.168.10.0/24, passerelle 192.168.10.1 (R0)
!   LAN1 (Switch1) : 192.168.20.0/24, passerelle 192.168.20.1 (R1)
!   PC0 : 192.168.20.10/24, passerelle 192.168.20.1
!
! ==================== Router0 ====================
enable
configure terminal
hostname Router0
no ip domain-lookup
enable secret A-CHANGER
!
interface GigabitEthernet0/0
 description Lien vers Router1
 ip address 192.168.12.1 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/1
 description LAN0 vers Switch0
 ip address 192.168.10.1 255.255.255.0
 no shutdown
!
! Route statique vers LAN1 via Router1
ip route 192.168.20.0 255.255.255.0 192.168.12.2
!
line console 0
 password A-CHANGER
 login
line vty 0 4
 password A-CHANGER
 login
end
write memory
!
! ==================== Router1 ====================
enable
configure terminal
hostname Router1
no ip domain-lookup
enable secret A-CHANGER
!
interface GigabitEthernet0/0
 description Lien vers Router0
 ip address 192.168.12.2 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/1
 description LAN1 vers Switch1
 ip address 192.168.20.1 255.255.255.0
 no shutdown
!
! Route statique vers LAN0 via Router0
ip route 192.168.10.0 255.255.255.0 192.168.12.1
!
! Route par defaut (exemple : vers le FAI / reseau externe, a adapter)
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
line console 0
 password A-CHANGER
 login
line vty 0 4
 password A-CHANGER
 login
end
write memory
!
! ==================== Switch0 / Switch1 (base) ====================
! Sur chaque commutateur : hostname Switch0 (ou Switch1), no ip domain-lookup,
! puis no shutdown sur les ports utilises :
!   Switch0 : Fa0/5 (vers Router0), Fa0/4 (vers Switch1)
!   Switch1 : Fa0/5 (vers Router1), Fa0/4 (vers Switch0), Fa0/1 (vers PC0)
!
! ==================== PC0 ====================
! Configuration IP statique (exemple) :
!   Adresse : 192.168.20.10, masque 255.255.255.0, passerelle 192.168.20.1
!
! ==================== Verification ====================
! Sur Router0 : show ip route | show ip route static
! Sur Router1 : show ip route | show ip route static
! Tests : ping 192.168.20.10 depuis Router0, ping 192.168.10.1 depuis PC0
! Test route par defaut : ping 203.0.113.1 depuis Router1 (si le FAI repond)
