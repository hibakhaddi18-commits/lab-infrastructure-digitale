! ============================================================
! Routeur R1 (LAN1) — RIPv2 + DHCPv4
! Affiché « Router2 » dans Packet Tracer.
! Extrait du show running-config (vérifié)
! Gig0/0/0 : 192.168.12.1/30 (liaison vers R2)
! Gig0/0/1 : 192.168.10.1/24 (LAN1)
! ============================================================
enable
configure terminal
! --- DHCP : exclusion puis pool ---
ip dhcp excluded-address 192.168.10.1 192.168.10.20
ip dhcp pool lan1
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
exit
! --- Interfaces ---
interface GigabitEthernet0/0/0
 ip address 192.168.12.1 255.255.255.252
exit
interface GigabitEthernet0/0/1
 ip address 192.168.10.1 255.255.255.0
exit
! --- Routage dynamique RIP version 2 ---
router rip
 version 2
 network 192.168.10.0
 network 192.168.12.0
 no auto-summary
exit
end
copy running-config startup-config
