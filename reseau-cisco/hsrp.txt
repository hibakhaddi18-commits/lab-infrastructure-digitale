! ============================================================
! FICHIER A ADAPTER - EXEMPLE DE CONFIGURATION
! Les adresses IP, noms et mots de passe ci-dessous sont des
! EXEMPLES tires de la topologie (capture d'ecran).
! Verifie et adapte-les a ton adressage reel avant usage.
! Ne pas appliquer tel quel sur un reseau de production.
! ============================================================
!
! Topologie : PC-A - S2 - R1, PC-B - S3 - R3, R1/R2/R3 interconnectes,
!             Router15, Server web
! HSRP sur le LAN des PC (S2 et S3 interconnectes) :
!   R1 = actif (priorite 110, preempt), R3 = secours (priorite 100)
! Adressage exemple :
!   LAN PC : 192.168.1.0/24
!   R1 (Gig0/1) : 192.168.1.2
!   R3 (Gig0/1) : 192.168.1.3
!   IP virtuelle HSRP (groupe 1) : 192.168.1.1 = passerelle des PC
!   MAC virtuelle du groupe 1 : 0000.0c07.ac01
!
! ==================== R1 (actif) ====================
enable
configure terminal
hostname R1
no ip domain-lookup
enable secret A-CHANGER
!
interface GigabitEthernet0/1
 description LAN PC vers S2
 ip address 192.168.1.2 255.255.255.0
 no shutdown
 standby 1 ip 192.168.1.1
 standby 1 priority 110
 standby 1 preempt
!
! Suivi du lien amont (optionnel, a adapter selon l'interface amont reelle) :
! standby 1 track GigabitEthernet0/0
!
! Routage amont vers le serveur web (exemple, a adapter) :
! ip route <reseau-serveur-web> <masque> <IP-R2>
end
write memory
!
! ==================== R3 (secours) ====================
enable
configure terminal
hostname R3
no ip domain-lookup
enable secret A-CHANGER
!
interface GigabitEthernet0/1
 description LAN PC vers S3
 ip address 192.168.1.3 255.255.255.0
 no shutdown
 standby 1 ip 192.168.1.1
 standby 1 priority 100
 standby 1 preempt
!
! Routage amont vers le serveur web (exemple, a adapter) :
! ip route <reseau-serveur-web> <masque> <IP-R2>
end
write memory
!
! ==================== PC-A / PC-B ====================
! Passerelle par defaut des PC : 192.168.1.1 (IP virtuelle HSRP)
! Les PC ne changent jamais de passerelle, meme en cas de bascule.
!
! ==================== Verification ====================
! Sur R1 et R3 : show standby | show standby brief
!   R1 doit afficher "Active", R3 "Standby".
! Test de bascule : sur R1, shutdown de GigabitEthernet0/1
!   -> R3 devient actif, les PC gardent la passerelle 192.168.1.1
!   -> ping depuis PC-A vers le serveur web doit continuer a repondre.
! Remettre ensuite l'interface en service : no shutdown (R1 reprend
! le role actif grace a preempt et a sa priorite 110).
