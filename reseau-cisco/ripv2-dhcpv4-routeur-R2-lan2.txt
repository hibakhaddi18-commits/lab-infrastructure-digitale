! ============================================================
! Routeur R2 — RIPv2 + DHCPv4
! Affiché « Router1 » dans Packet Tracer.
! Hostname, DHCP et interfaces vérifiés (show running-config
! et show ip interface brief).
! RIP vérifié par le test : R2 apprend 192.168.10.0/24 via
! 192.168.12.1 et R1 apprend 192.168.20.0/24 via 192.168.12.2.
! Gig0/0/0 : 192.168.12.2/30 (liaison vers R1)
! Gig0/0/1 : 192.168.20.1/24 (LAN2)
! ============================================================
enable
configure terminal
hostname R2
! --- DHCP : exclusion puis pool ---
ip dhcp excluded-address 192.168.20.1 192.168.20.20
ip dhcp pool lan2
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8
exit
! --- Interfaces ---
interface GigabitEthernet0/0/0
 ip address 192.168.12.2 255.255.255.252
exit
interface GigabitEthernet0/0/1
 ip address 192.168.20.1 255.255.255.0
exit
! --- Routage dynamique RIP version 2 ---
router rip
 version 2
 network 192.168.20.0
 network 192.168.12.0
 no auto-summary
exit
end
copy running-config startup-config
